# Additional role-level targets for the four-country mobile transition.
# GER keeps its vanilla motorized and mechanized_assault role entries.
# USA keeps its vanilla mobile role entry.

hof_motorized_ENG_ITA = {
	available_for = { ENG ITA }
	role = motorized
	upgrade_prio = {
		base = 3
		modifier = { factor = 0 NOT = { has_tech = motorised_infantry } }
	}
	hof_motorized_early = {
		upgrade_prio = { base = 2 }
		target_template = {
			support = { engineer = 1 artillery = 1 }
			regiments = { motorized = 9 }
		}
	}
	hof_motorized_late = {
		upgrade_prio = {
			base = 1
			modifier = { factor = 4 date > 1939.12.31 }
		}
		target_template = {
			support = { engineer = 1 artillery = 1 logistics_company = 1 }
			regiments = { motorized = 9 mot_artillery_brigade = 1 }
		}
	}
}
