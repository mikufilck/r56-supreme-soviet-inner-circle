# ---- 头：莫洛托夫计划（唯一的 shared_focus）----
shared_focus = {
	id = hof_ma_founding
	icon = GFX_focus_SOV_the_path_of_marxism_leninism
	x = 0
	y = 27
	offset = { x = 0 y = 60 trigger = { tag = TAN } }
	cost = 10

	# 普通树 y=27..33；图瓦单独 y=87..93

	# 准入（用户口径：共产主义政体 且 与苏联同一阵营或为其附庸 —— 与北欧一样「符合条件大家都能点」）
	# ⚠️ 用 available 而不是 allow_branch：allow_branch 只在国策树加载时判定，
	#    一个国家【中途】转成共产主义时不一定刷新得到；available 是持续判定的，
	#    而且正好给出用户要的「看得见、点不了」。
	available = {
		custom_trigger_tooltip = {
			tooltip = hof_ma_available_government_tt
			has_government = communism
		}
		custom_trigger_tooltip = {
			tooltip = hof_ma_available_tt
			OR = {
				tag = SOV
				is_in_faction_with = SOV
				is_subject_of = SOV
			}
		}
	}

	search_filters = { FOCUS_FILTER_POLITICAL }

	completion_reward = {
		hidden_effect = { SOV = { set_country_flag = hof_ma_done_hof_ma_founding } }
		if = {
			limit = { hof_ma_eligible = yes }
		custom_effect_tooltip = hof_ma_head_dm_tt
		# 累积数值：三个容器一起喂
		add_to_variable = {
			hof_ma_eco_trade = 0.05
			tooltip = hof_ma_tt_faction_trade_opinion_factor
		}
		add_to_variable = {
			hof_ma_pol_opinion = 0.05
			tooltip = hof_ic_tt_opinion_gain_monthly_factor
		}
		add_to_variable = {
			hof_ma_mil_command_max = 25
			tooltip = max_command_power_tt
		}
		# 三个动态民族精神一起挂上 —— 头节点是唯一的入口
		add_dynamic_modifier = { modifier = hof_ma_dm_eco }
		add_dynamic_modifier = { modifier = hof_ma_dm_pol }
		add_dynamic_modifier = { modifier = hof_ma_dm_mil }
		set_country_flag = hof_ma_joined
		add_political_power = 50
		}
		hidden_effect = { SOV = { hof_ma_reconcile_all = yes } }
	}
}
